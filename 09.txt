    function getGeoJSONBounds(geojson) {
      const bounds = new maplibregl.LngLatBounds();
      geojson.features.forEach((feature) => {
        const geom = feature.geometry;
        if (!geom) return;
        if (geom.type === 'Point') {
          bounds.extend(geom.coordinates);
        } else if (geom.type === 'LineString' || geom.type === 'MultiPoint') {
          geom.coordinates.forEach((coord) => bounds.extend(coord));
        } else if (geom.type === 'Polygon' || geom.type === 'MultiLineString') {
          geom.coordinates.forEach((ring) => ring.forEach((coord) => bounds.extend(coord)));
        } else if (geom.type === 'MultiPolygon') {
          geom.coordinates.forEach((poly) =>
            poly.forEach((ring) => ring.forEach((coord) => bounds.extend(coord)))
          );
        }
      });
      return bounds;
    }

    // Stato Course-Up disattivato di default all'apertura
    let courseUpEnabled = false;
    let navMarker = null;
    let lastHeading = 0;
    let lastCoords = null;
    let isProgrammaticMove = false;

    function moveCameraSafe(options) {
      isProgrammaticMove = true;
      map.easeTo(options);
      map.once('moveend', () => {
        isProgrammaticMove = false;
      });
    }

    async function requestCompassPermissionIfNeeded() {
      if (
        typeof DeviceOrientationEvent !== 'undefined' &&
        typeof DeviceOrientationEvent.requestPermission === 'function'
      ) {
        try {
          await DeviceOrientationEvent.requestPermission();
        } catch (_) {}
      }
    }

    // Controllo per passare da 2D a 3D mantenendo lo stato Course-Up
    class ViewModeControl {
      onAdd(map) {
        this._map = map;
        this._container = document.createElement('div');
        this._container.className = 'maplibregl-ctrl';

        this._btn = document.createElement('button');
        this._btn.type = 'button';
        this._btn.className = 'view-mode-btn';
        this._btn.textContent = '2D';
        this._btn.title = 'Passa tra vista 2D e vista 3D';

        this._btn.addEventListener('click', () => {
          is3DMode = !is3DMode;
          this.updateUI();

          const cameraOptions = {
            pitch: getTargetPitch(),
            duration: 450
          };

          if (courseUpEnabled && lastCoords) {
            cameraOptions.center = lastCoords;
            cameraOptions.bearing = lastHeading;
          }

          moveCameraSafe(cameraOptions);
        });

        this._container.appendChild(this._btn);
        return this._container;
      }

      updateUI() {
        if (is3DMode) {
          this._btn.textContent = '3D';
          this._btn.classList.add('is-3d');
        } else {
          this._btn.textContent = '2D';
          this._btn.classList.remove('is-3d');
        }
      }

      onRemove() {
        this._container.parentNode.removeChild(this._container);
        this._map = undefined;
      }
    }

    const viewModeCtrl = new ViewModeControl();
    map.addControl(viewModeCtrl, 'top-right');

    // Controllo personalizzato per attivare/disattivare Course-Up
    class CourseUpControl {
      onAdd(map) {
        this._map = map;
        this._container = document.createElement('div');
        this._container.className = 'maplibregl-ctrl';

        this._btn = document.createElement('button');
        this._btn.type = 'button';
        this._btn.className = 'course-up-btn';
        this._btn.innerHTML = '🧭 Course-Up';
        this._btn.title = 'Attiva/Disattiva rotazione automatica mappa';

        this._btn.addEventListener('click', async () => {
          courseUpEnabled = !courseUpEnabled;
          this.updateUI();

          if (courseUpEnabled) {
            await requestCompassPermissionIfNeeded();
            if (lastCoords) {
              moveCameraSafe({
                center: lastCoords,
                bearing: lastHeading,
                pitch: getTargetPitch(),
                duration: 450
              });
            }
          } else {
            moveCameraSafe({
              bearing: 0,
              pitch: getTargetPitch(),
              duration: 450
            });
          }
        });

        this._container.appendChild(this._btn);
        return this._container;
      }

      updateUI() {
        if (courseUpEnabled) {
          this._btn.classList.add('active');
        } else {
          this._btn.classList.remove('active');
        }
      }

      onRemove() {
        this._container.parentNode.removeChild(this._container);
        this._map = undefined;
      }
    }

    const courseUpCtrl = new CourseUpControl();
    map.addControl(courseUpCtrl, 'top-right');

    // Se l'utente sposta o ruota manualmente la mappa con le dita, mette in pausa Course-Up
    map.on('dragstart', () => {
      if (!isProgrammaticMove && courseUpEnabled) {
        courseUpEnabled = false;
        courseUpCtrl.updateUI();
      }
    });

    function updateNavigationView(coords, heading) {
      if (!Array.isArray(coords) || coords.length !== 2) return;
      const [lon, lat] = coords;
      if (typeof lon !== 'number' || typeof lat !== 'number' || Number.isNaN(lon) || Number.isNaN(lat)) return;

      lastCoords = [lon, lat];

      if (typeof heading === 'number' && !Number.isNaN(heading)) {
        lastHeading = ((heading % 360) + 360) % 360;
      }

      if (!navMarker) {
        navMarker = new maplibregl.Marker({
          element: wrapperEl,
          rotationAlignment: 'map',
          pitchAlignment: 'map'
        })
          .setLngLat(lastCoords)
          .setRotation(lastHeading)
          .addTo(map);
      } else {
        navMarker.setLngLat(lastCoords);
        navMarker.setRotation(lastHeading);
      }

      // Sposta la camera solo se l'utente ha attivato Course-Up
      if (courseUpEnabled) {
        moveCameraSafe({
          center: lastCoords,
          bearing: lastHeading,
          pitch: getTargetPitch(),
          duration: 300,
          easing: (t) => t
        });
      }
    }

    geolocate.on('geolocate', (e) => {
      const { longitude, latitude, heading } = e.coords;
      updateNavigationView([longitude, latitude], heading);
    });

    // Tracciamento GPS in background: aggiorna la freccia senza spostare l'inquadratura iniziale
    function startBackgroundTracking() {
      if (!('geolocation' in navigator)) return;
      navigator.geolocation.watchPosition(
        (pos) => {
          const { longitude, latitude, heading } = pos.coords;
          updateNavigationView([longitude, latitude], heading);
        },
        () => {},
        {
          enableHighAccuracy: true,
          maximumAge: 5000,
          timeout: 10000
        }
      );
    }

    window.addEventListener('deviceorientationabsolute', (e) => {
      if (typeof e.alpha === 'number' && !Number.isNaN(e.alpha) && lastCoords && courseUpEnabled) {
        const compassHeading = (360 - e.alpha) % 360;
        updateNavigationView(lastCoords, compassHeading);
      }
    });

    window.addEventListener('deviceorientation', (e) => {
      if (typeof e.webkitCompassHeading === 'number' && !Number.isNaN(e.webkitCompassHeading) && lastCoords && courseUpEnabled) {
        updateNavigationView(lastCoords, e.webkitCompassHeading);
      }
    });

    map.on('load', () => {
      // Aggiunge la sorgente dati e il layer grafico del confine Territorio 09
      map.addSource('territorio-09', {
        type: 'geojson',
        data: territorio09GeoJSON

        paint: {
          'line-color': '#0066ff',
          'line-width': 6,
          'line-opacity': 1
        }
      });

      if (!territoryBounds.isEmpty()) {
        isProgrammaticMove = true;
        map.fitBounds(territoryBounds, {
          padding: 110,
          maxZoom: 15,
          duration: 0
        });
        map.once('moveend', () => {
          isProgrammaticMove = false;
        });
      }

      startBackgroundTracking();
    });
